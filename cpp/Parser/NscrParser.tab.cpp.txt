// Static helper functions implementation
static std::vector<TYArgment*>* arrayIfNotArray(TYArgment* receiver)
{
    if (!receiver || !receiver->canAddObject()) {
        std::vector<TYArgment*>* arr = new std::vector<TYArgment*>();
        arr->push_back(receiver);
        return arr;
    }
    return receiver->toArray();
}

static std::vector<TYArgment*>* arrayWithAddObject(TYArgment* receiver, TYArgment* aObject)
{
    std::vector<TYArgment*>* arr = nullptr;

    if (!receiver || !receiver->canAddObject()) {
        arr = new std::vector<TYArgment*>();
        arr->push_back(receiver);
    } else {
        arr = receiver->toArray();
    }

    if (aObject && aObject->canAddObject()) {
        std::vector<TYArgment*>* srcArr = aObject->toArray();
        arr->insert(arr->end(), srcArr->begin(), srcArr->end());
    } else {
        arr->push_back(aObject);
    }

    return arr;
}

static std::vector<TYArgment*>* arrayOfForCmd(TYArgment* bindVar,
                                              TYOpType op,
                                              TYArgment* start,
                                              TYArgment* to_token,
                                              TYArgment* end,
                                              TYArgment* step_token,
                                              TYArgment* step)
{
    if (op != TYEqualOpType) {
        return nullptr;
    }

    std::string toStr = to_token ? to_token->stringObj() : "";
    if (toStr != "to") {
        return nullptr;
    }

    if (step_token) {
        std::string stepStr = step_token->stringObj();
        if (stepStr != "step") {
            return nullptr;
        }
    }

    if (!step) {
        step = TYArgment::createIntNumber(1);
    }

    std::vector<TYArgment*>* arr = new std::vector<TYArgment*>();
    arr->push_back(bindVar);
    arr->push_back(start);
    arr->push_back(end);
    arr->push_back(step);

    return arr;
}
